# Cineby crawl policy (RFC 9309).
#
# /api/tmdb, /api/discover and /api/segments stay crawlable on purpose: public pages load rows
# from them, and blocking them would stop Googlebot from rendering that content.
# Account pages (/history, /watchlist, /profiles, /settings) are not listed here because they
# carry a noindex tag, and crawlers must be able to fetch a page to see it.

User-agent: *
Content-Signal: search=yes, ai-input=yes, ai-train=no
Allow: /
Disallow: /watch/
Disallow: /api/auth/
Disallow: /api/stream/
Disallow: /api/history
Disallow: /api/profiles
Disallow: /api/watchlist
Disallow: /api/settings

# AI search crawlers: allowed so Cineby can appear in AI-assistant answers.
User-agent: OAI-SearchBot
User-agent: Claude-SearchBot
User-agent: PerplexityBot
Content-Signal: search=yes, ai-input=yes, ai-train=no
Allow: /
Disallow: /watch/
Disallow: /api/auth/
Disallow: /api/stream/
Disallow: /api/history
Disallow: /api/profiles
Disallow: /api/watchlist
Disallow: /api/settings

# AI training crawlers: blocked. This does not affect appearing in AI search results.
User-agent: GPTBot
User-agent: ClaudeBot
User-agent: Google-Extended
User-agent: Applebot-Extended
User-agent: CCBot
User-agent: Bytespider
User-agent: Amazonbot
Disallow: /

Sitemap: https://cineby.group/sitemap.xml
